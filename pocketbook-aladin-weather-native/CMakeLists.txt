cmake_minimum_required(VERSION 3.10)
project(HelcinoPocasi C)
set(CMAKE_C_STANDARD 99)
find_path(INKVIEW_INCLUDE_DIR inkview.h HINTS "$ENV{SDK_BASE}/usr/arm-obreey-linux-gnueabi/sysroot/usr/local/include")
find_library(INKVIEW_LIBRARY inkview HINTS "$ENV{SDK_BASE}/usr/arm-obreey-linux-gnueabi/sysroot/usr/local/lib")
find_path(FREETYPE_INCLUDE_DIR ft2build.h HINTS "$ENV{SDK_BASE}/usr/arm-obreey-linux-gnueabi/sysroot/usr/include/freetype2")
if(NOT INKVIEW_INCLUDE_DIR OR NOT INKVIEW_LIBRARY)
  message(FATAL_ERROR "PocketBook InkView SDK not found")
endif()
add_executable(HelcinoPocasi.app src/main.c)
target_include_directories(HelcinoPocasi.app PRIVATE "${INKVIEW_INCLUDE_DIR}" "${FREETYPE_INCLUDE_DIR}")
target_link_libraries(HelcinoPocasi.app PRIVATE "${INKVIEW_LIBRARY}" m)
target_compile_definitions(HelcinoPocasi.app PRIVATE _GNU_SOURCE)
target_compile_options(HelcinoPocasi.app PRIVATE -Wall -Wextra -Wno-unused-parameter)
